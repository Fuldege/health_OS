// ======================================== // Jing Health OS - Apps Script API // ======================================== const SHEETS = ['Appetite_Log', 'Medical_Log', 'Body_Log', 'Hormone_Log', 'Sleep_Log', 'Food_Log', 'Beauty_Log', 'Daily_Status']; const PRIVATE_SHEETS = ['Medical_Log', 'Hormone_Log']; const PRIVATE_PWD = 'jing'; const PHOTO_FIELD_MAP = { 'Beauty_Log': 'hair_part_photo_url', 'Body_Log': 'photo_url' }; const PHOTO_FOLDER_NAME = 'Jing Health OS Photos'; function doGet(e) { try { const params = e.parameter; const sheetName = params.sheet || 'Appetite_Log'; if (sheetName === 'dashboard') { return jsonOutput(buildDashboard()); } if (!SHEETS.includes(sheetName)) { return jsonOutput({ error: '未知的 sheet: ' + sheetName }); } if (PRIVATE_SHEETS.includes(sheetName) && params.pwd !== PRIVATE_PWD) { return jsonOutput({ error: '密码错误，无权访问该表' }); } let rows = readSheet(sheetName); if (params.range && /^(\d+)d$/.test(params.range)) { const days = parseInt(params.range.match(/^(\d+)d$/)[1], 10); const cutoff = new Date(); cutoff.setDate(cutoff.getDate() - days); rows = rows.filter(r => r.date && new Date(r.date) >= cutoff); } return jsonOutput(rows); } catch (err) { return jsonOutput({ error: err.message }); } } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const sheetName = body.sheet; const data = body.data || {}; if (!SHEETS.includes(sheetName)) { return jsonOutput({ error: '未知的 sheet: ' + sheetName }); } if (PRIVATE_SHEETS.includes(sheetName) && body.pwd !== PRIVATE_PWD) { return jsonOutput({ error: '密码错误，无权写入该表' }); } const result = appendRow(sheetName, data); return jsonOutput({ success: true, row: result }); } catch (err) { return jsonOutput({ error: err.message }); } } // ---------- 核心读写 ---------- function getSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(name); if (!sheet) throw new Error('找不到 sheet: ' + name); return sheet; } function readSheet(name) { const sheet = getSheet(name); const values = sheet.getDataRange().getDisplayValues(); if (values.length < 2) return []; const headers = values[0]; const rows = []; for (let i = 1; i < values.length; i++) { const row = values[i]; if (row.every(c => c === '')) continue; const obj = {}; headers.forEach((h, idx) => { obj[h] = row[idx]; }); rows.push(obj); } return rows; } function appendRow(name, data) { const sheet = getSheet(name); const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; // 图片上传：base64 转存 Drive，只把链接写进表格 if (data.photo_base64) { const field = PHOTO_FIELD_MAP[name]; if (field) { data[field] = savePhotoAndGetUrl(data.photo_base64); } delete data.photo_base64; } if (headers.includes('id') && !data.id) { data.id = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMddHHmmss'); } if (headers.includes('date') && !data.date) { data.date = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); } if (name === 'Appetite_Log' && !data.injection_day_offset) { data.injection_day_offset = computeInjectionDayOffset(data.date); } if (name === 'Medical_Log' && !data.cycle_index) { data.cycle_index = countMedicalEntriesBefore(data.date) + 1; } if (name === 'Body_Log' && !data.avg_weight_7d) { data.avg_weight_7d = computeAvgWeight7d(data.date, data.weight_kg); } const row = headers.map(h => { let v = data[h] !== undefined ? data[h] : ''; if ((h === 'date' || h === 'time') && v !== '') v = "'" + v; return v; }); sheet.appendRow(row); return data; } // ---------- 图片存储 ---------- function getOrCreatePhotoFolder() { const folders = DriveApp.getFoldersByName(PHOTO_FOLDER_NAME); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(PHOTO_FOLDER_NAME); } function savePhotoAndGetUrl(base64DataUrl) { const matches = base64DataUrl.match(/^data:(image\/\w+);base64,(.+)$/); if (!matches) throw new Error('图片格式不正确'); const mimeType = matches[1]; const base64Data = matches[2]; const bytes = Utilities.base64Decode(base64Data); const blob = Utilities.newBlob(bytes, mimeType, 'photo_' + new Date().getTime()); const folder = getOrCreatePhotoFolder(); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return 'https://drive.google.com/uc?export=view&id=' + file.getId(); } // ---------- 特殊计算 ---------- function computeInjectionDayOffset(dateStr) { const medRows = readSheet('Medical_Log'); if (medRows.length === 0) return ''; const target = new Date(dateStr); let latest = null; medRows.forEach(r => { const d = new Date(r.date); if (d <= target && (!latest || d > latest)) latest = d; }); if (!latest) return ''; return Math.round((target - latest) / (1000 * 60 * 60 * 24)); } function countMedicalEntriesBefore(dateStr) { const medRows = readSheet('Medical_Log'); const target = new Date(dateStr); return medRows.filter(r => new Date(r.date) <= target).length; } function computeAvgWeight7d(dateStr, currentWeight) { const bodyRows = readSheet('Body_Log'); const target = new Date(dateStr); const cutoff = new Date(target); cutoff.setDate(cutoff.getDate() - 6); const recent = bodyRows .filter(r => { const d = new Date(r.date); return d >= cutoff && d <= target && r.weight_kg; }) .map(r => Number(r.weight_kg)); recent.push(Number(currentWeight)); const avg = recent.reduce((a, b) => a + b, 0) / recent.length; return Math.round(avg * 100) / 100; } // ---------- Dashboard 汇总 ---------- function buildDashboard() { const body = readSheet('Body_Log'); const sleep = readSheet('Sleep_Log'); const appetite = readSheet('Appetite_Log'); const beauty = readSheet('Beauty_Log'); const medical = readSheet('Medical_Log'); const today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); const lastBody = body.length ? body[body.length - 1] : {}; const lastSleep = sleep.filter(r => r.date === today); const todayAppetite = appetite.filter(r => r.date === today); const lastBeauty = beauty.length ? beauty[beauty.length - 1] : {}; const lastMedical = medical.length ? medical[medical.length - 1] : {}; let appetitePeak = null; todayAppetite.forEach(r => { if (!appetitePeak || Number(r.intensity_0_10) > Number(appetitePeak.intensity_0_10)) { appetitePeak = r; } }); return { date: today, weight_kg: lastBody.weight_kg || null, avg_weight_7d: lastBody.avg_weight_7d || null, sleep_hours: lastSleep.length ? lastSleep[0].sleep_hours : null, appetite_peak_0_10: appetitePeak ? appetitePeak.intensity_0_10 : null, appetite_main_trigger: appetitePeak ? appetitePeak.trigger : null, injection_day_offset: lastMedical.date ? computeInjectionDayOffset(today) : null, beauty_trend_note: lastBeauty.skin_state_0_10 ? `皮肤 ${lastBeauty.skin_state_0_10}/10` : null }; } // ---------- 输出 ---------- function jsonOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
